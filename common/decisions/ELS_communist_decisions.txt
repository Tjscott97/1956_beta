# Unlocked by Rebuild the Party Cells. Grows communist support while the party is still underground,
# the category hides once the communists are in power.

ELS_clandestine_network = {

	ELS_print_clandestine_newspapers = {

		icon = generic_propaganda

		allowed = {
			original_tag = ELS
		}

		cost = 25

		fire_only_once = no
		days_re_enable = 60

		ai_will_do = {
			factor = 1
		}

		complete_effect = {
			add_popularity = {
				ideology = communism
				popularity = 0.03
			}
		}
	}

	ELS_organize_workers_study_circles = {

		icon = generic_political_rally

		allowed = {
			original_tag = ELS
		}

		cost = 40

		fire_only_once = no
		days_re_enable = 90

		ai_will_do = {
			factor = 1
		}

		# Bigger gain than the newspapers, but the Guardia Nacional notices the meetings
		complete_effect = {
			add_popularity = {
				ideology = communism
				popularity = 0.05
			}
			add_stability = -0.02
		}
	}

	ELS_contact_socorro_rojo = {

		icon = generic_civil_support

		allowed = {
			original_tag = ELS
		}

		cost = 50

		fire_only_once = yes

		ai_will_do = {
			factor = 1
		}

		# International Red Aid, the Comintern's support network for imprisoned and exiled communists
		complete_effect = {
			add_popularity = {
				ideology = communism
				popularity = 0.05
			}
			custom_effect_tooltip = generic_skip_one_line_tt

			custom_effect_tooltip = ELS_modify_coffee_tt
			add_to_variable = { ELS_coffee_communism_drift = 0.01 tooltip = communism_drift_tt }
		}
	}
}

# Unlocked by Fund the Revolutionary Cells. One decision targeting each Central American neighbour (FROM is the target).
# The civilian factory is only used while the decision is active (90 days), the same length as the idea it gives.

ELS_export_the_revolution = {

	ELS_fund_communist_cells = {

		icon = generic_prepare_civil_war

		allowed = {
			original_tag = ELS
		}

		targets = { GUA HON NIC COS }

		target_trigger = {
			FROM = { exists = yes }
			NOT = { has_war_with = FROM }
		}

		available = {
			FROM = {
				NOT = { has_government = communism }
				NOT = { has_idea = r56_ELS_funded_communist_cells }
			}
		}

		fire_only_once = no
		days_remove = 90

		custom_cost_trigger = {
			has_equipment = { infantry_equipment > 99 }
		}
		custom_cost_text = decision_cost_ie_100

		modifier = {
			civilian_factory_use = 1
		}

		ai_will_do = {
			factor = 1
		}

		complete_effect = {
			add_equipment_to_stockpile = {
				type = infantry_equipment
				amount = -100
			}
			FROM = {
				add_timed_idea = {
					idea = r56_ELS_funded_communist_cells
					days = 90
				}
			}
		}
	}
}
