# Unlocked by Rebuild the Party Cells. Grows communist support while the party is still underground,
# the category hides once the communists are in power.

ELS_clandestine_network = {

	ELS_print_clandestine_newspapers = {

		icon = generic_propaganda

		allowed = {
			original_tag = ELS
		}

		cost = 25

		fire_only_once = no
		days_re_enable = 60

		ai_will_do = {
			factor = 1
		}

		complete_effect = {
			add_popularity = {
				ideology = communism
				popularity = 0.03
			}
		}
	}

	ELS_organize_workers_study_circles = {

		icon = generic_political_rally

		allowed = {
			original_tag = ELS
		}

		cost = 40

		fire_only_once = no
		days_re_enable = 90

		ai_will_do = {
			factor = 1
		}

		# Bigger gain than the newspapers, but the Guardia Nacional notices the meetings
		complete_effect = {
			add_popularity = {
				ideology = communism
				popularity = 0.05
			}
			add_stability = -0.02
		}
	}

	ELS_contact_socorro_rojo = {

		icon = generic_civil_support

		allowed = {
			original_tag = ELS
		}

		cost = 50

		fire_only_once = yes

		ai_will_do = {
			factor = 1
		}

		# International Red Aid, the Comintern's support network for imprisoned and exiled communists
		complete_effect = {
			add_popularity = {
				ideology = communism
				popularity = 0.05
			}
			custom_effect_tooltip = generic_skip_one_line_tt

			custom_effect_tooltip = ELS_modify_coffee_tt
			add_to_variable = { ELS_coffee_communism_drift = 0.01 tooltip = communism_drift_tt }
		}
	}
}
