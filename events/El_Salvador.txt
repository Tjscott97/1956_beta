country_event = {  #Joint Army Exercises
    id = el_salvador.1
    title = el_salvador.1.t
    desc = el_salvador.1.d
    picture = GFX_news_event_generic 
    is_triggered_only = yes 

    option = {
        name = el_salvador.1.a # "Accept"

        # decline weight is 15:
        # opinion 0 or less ~12%, above 0 40%, above 15 ~63%, above 30 ~77%, above 50 ~87%, above 75 ~93%
        ai_chance = {
            base = 2
            modifier = {
                add = 8
                has_opinion = { target = FROM value > 0 }
            }
            modifier = {
                add = 15
                has_opinion = { target = FROM value > 15 }
            }
            modifier = {
                add = 25
                has_opinion = { target = FROM value > 30 }
            }
            modifier = {
                add = 50
                has_opinion = { target = FROM value > 50 }
            }
            modifier = {
                add = 100
                has_opinion = { target = FROM value > 75 }
            }
            modifier = {
                factor = 1.5
                is_in_faction_with = FROM
            }
        }

        # Target gets army XP
        army_experience = 20.0

        # Target improves opinion of El Salvador
        ROOT = {
            add_opinion_modifier = {
                target = FROM
                modifier = joint_army_exercises_opinion
                days = 360
            }
        }

        hidden_effect = { ELS_offer_accepted_response = yes }
    }

    option = {
        name = el_salvador.1.b # "Decline"

        ai_chance = {
            base = 15
        }

        hidden_effect = { ELS_offer_declined_response = yes }
    }
}
